\definecolor{pGreen}{HTML}{663366}
\pgfdeclarelindenmayersystem{Penrose}{
\symbol{F}{\pgflsystemdrawforward}
\symbol{+}{\pgflsystemturnright} % Explicitly define + and - symbols.
\symbol{-}{\pgflsystemturnleft}
\rule{W -> YF++ZF----XF[-YF----WF]++}
\rule{X -> +YF--ZF[---WF--XF]+}
\rule{Y -> -WF++XF[+++YF++ZF]-}
\rule{Z -> --YF++++WF[+ZF++++XF]--XF}
}
\begin{document}
\begin{tikzpicture}
      \draw[lindenmayer system={Penrose, step=512, axiom=[X]++[X]++[X]++[X]++[X], order=4, angle=36},pGreen] lindenmayer system;
